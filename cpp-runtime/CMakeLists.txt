cmake_minimum_required(VERSION 3.16)
project(logistics_cpp_runtime VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
add_library(runtime_domain src/domain.cpp)
target_include_directories(runtime_domain PUBLIC src)
add_library(runtime_catalog src/catalog.cpp src/point_audit.cpp)
target_include_directories(runtime_catalog PUBLIC src)
add_library(runtime_core src/core.cpp src/adapters.cpp src/control.cpp)
target_include_directories(runtime_core PUBLIC src)
target_link_libraries(runtime_core PUBLIC runtime_domain SQLite::SQLite3 Threads::Threads)
if(MSVC)
  target_compile_options(runtime_core PUBLIC /W4 /utf-8)
  target_compile_options(runtime_catalog PUBLIC /W4 /utf-8)
else()
  target_compile_options(runtime_core PUBLIC -Wall -Wextra -Wpedantic)
  target_compile_options(runtime_catalog PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(collector src/collector.cpp src/supervisor.cpp)
target_link_libraries(collector PRIVATE runtime_core)
add_executable(diagnose src/diagnose.cpp src/snapshot_json.cpp)
target_link_libraries(diagnose PRIVATE runtime_core)
add_executable(simulation-archive src/archive.cpp)
target_link_libraries(simulation-archive PRIVATE SQLite::SQLite3 Threads::Threads)
option(BUILD_MCP "Build the C++ stdio MCP diagnostics" ON)
option(BUILD_MCP_HTTP "Build the opt-in Linux loopback MCP HTTP adapter" OFF)
if(BUILD_MCP)
  find_package(nlohmann_json 3.11 CONFIG REQUIRED)
  add_library(runtime_mcp src/mcp.cpp)
  target_link_libraries(runtime_mcp PUBLIC runtime_core nlohmann_json::nlohmann_json)
  add_executable(logistics-mcp src/mcp_stdio.cpp)
  target_link_libraries(logistics-mcp PRIVATE runtime_mcp)
endif()
if(BUILD_MCP_HTTP)
  if(NOT BUILD_MCP OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "BUILD_MCP_HTTP requires BUILD_MCP and Linux")
  endif()
  find_package(OpenSSL 3 COMPONENTS Crypto REQUIRED)
  find_path(MICROHTTPD_INCLUDE_DIR microhttpd.h REQUIRED)
  find_library(MICROHTTPD_LIBRARY microhttpd REQUIRED)
  add_executable(logistics-mcp-http src/mcp_http.cpp)
  target_include_directories(logistics-mcp-http PRIVATE ${MICROHTTPD_INCLUDE_DIR})
  target_link_libraries(logistics-mcp-http PRIVATE runtime_mcp OpenSSL::Crypto ${MICROHTTPD_LIBRARY})
endif()
option(BUILD_HMI "Build independent Qt Widgets HMI" ON)
if(BUILD_HMI)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
  add_library(runtime_widgets src/workbench.cpp src/workbench.hpp src/schematic.cpp src/schematic.hpp)
  set_target_properties(runtime_widgets PROPERTIES AUTOMOC ON)
  target_link_libraries(runtime_widgets PUBLIC runtime_core runtime_catalog Qt6::Widgets)
  add_executable(hmi src/hmi.cpp)
  target_link_libraries(hmi PRIVATE runtime_widgets)
  if(WIN32)
    add_executable(Logistics WIN32 deploy/windows/launcher.cpp)
    target_link_libraries(Logistics PRIVATE user32)
    if(MINGW)
      target_link_options(Logistics PRIVATE -municode)
    endif()
  endif()
endif()
include(CTest)
option(BUILD_OPCUA_LOOPBACK_FIXTURE "Build the isolated read-only open62541 test" OFF)
if(BUILD_TESTING)
  add_executable(domain_test tests/domain.cpp)
  target_link_libraries(domain_test PRIVATE runtime_domain)
  add_test(NAME domain COMMAND domain_test)
  add_executable(catalog_test tests/catalog.cpp)
  target_link_libraries(catalog_test PRIVATE runtime_catalog)
  add_test(NAME catalog COMMAND catalog_test)
  add_executable(point_audit_test tests/point_audit.cpp)
  target_link_libraries(point_audit_test PRIVATE runtime_catalog)
  add_test(NAME point_audit COMMAND point_audit_test)
  add_executable(control_test_client tests/control_client.cpp)
  target_link_libraries(control_test_client PRIVATE runtime_core)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  if(BUILD_MCP)
    add_test(NAME mcp_stdio COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/mcp_stdio.py $<TARGET_FILE:collector> $<TARGET_FILE:logistics-mcp> $<TARGET_FILE:diagnose>)
    set_tests_properties(mcp_stdio PROPERTIES TIMEOUT 35)
  endif()
  if(BUILD_MCP_HTTP)
    add_test(NAME mcp_http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/mcp_http.py $<TARGET_FILE:collector> $<TARGET_FILE:logistics-mcp-http>)
    set_tests_properties(mcp_http PROPERTIES TIMEOUT 35)
    find_program(NGINX_EXECUTABLE nginx)
    find_program(OPENSSL_EXECUTABLE openssl)
    if(NGINX_EXECUTABLE AND OPENSSL_EXECUTABLE)
      add_test(NAME mcp_https_gateway COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/mcp_http.py $<TARGET_FILE:collector> $<TARGET_FILE:logistics-mcp-http> ${NGINX_EXECUTABLE} ${OPENSSL_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/deploy/nginx-mcp.conf.in)
      set_tests_properties(mcp_https_gateway PROPERTIES TIMEOUT 45)
    else()
      message(STATUS "nginx or openssl executable unavailable: MCP HTTPS gateway CTest will not be registered")
    endif()
  endif()
  if(BUILD_OPCUA_LOOPBACK_FIXTURE)
    find_package(open62541 1.4.11 EXACT CONFIG REQUIRED)
    add_executable(opcua_loopback_fixture tests/opcua_loopback.cpp)
    target_link_libraries(opcua_loopback_fixture PRIVATE open62541::open62541 Threads::Threads)
    add_test(NAME opcua_loopback COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/opcua_loopback.py $<TARGET_FILE:opcua_loopback_fixture>)
    find_program(OPENSSL_EXECUTABLE openssl)
    if(OPENSSL_EXECUTABLE)
      add_test(NAME opcua_secure_loopback COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/opcua_loopback.py $<TARGET_FILE:opcua_loopback_fixture> ${OPENSSL_EXECUTABLE})
      set_tests_properties(opcua_secure_loopback PROPERTIES TIMEOUT 30)
    endif()
  endif()
  add_test(NAME modes COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/modes.py $<TARGET_FILE:collector> $<TARGET_FILE:control_test_client> $<TARGET_FILE:diagnose>)
  add_test(NAME acceptance COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/acceptance.py $<TARGET_FILE:collector> $<TARGET_FILE:diagnose>)
  add_test(NAME native_migration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/native_migration.py $<TARGET_FILE:collector> $<TARGET_FILE:diagnose> $<TARGET_FILE:control_test_client>)
  add_test(NAME archive COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/archive.py $<TARGET_FILE:collector> $<TARGET_FILE:diagnose> $<TARGET_FILE:simulation-archive>)
  set_tests_properties(archive PROPERTIES TIMEOUT 35)
  add_test(NAME logs COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/logs.py $<TARGET_FILE:collector> $<TARGET_FILE:control_test_client>)
  if(BUILD_HMI)
    add_executable(widgets_test tests/widgets.cpp)
    target_link_libraries(widgets_test PRIVATE runtime_widgets)
    add_test(NAME widgets_controls COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/widgets.py $<TARGET_FILE:collector> $<TARGET_FILE:widgets_test> $<TARGET_FILE:hmi> $<TARGET_FILE:control_test_client>)
    set_tests_properties(widgets_controls PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_test(NAME hmi_refresh COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/hmi.py $<TARGET_FILE:collector> $<TARGET_FILE:hmi>)
    set_tests_properties(hmi_refresh PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
endif()
